\lstset{style=cstyle}

% ---------- Eigene Befehle ----------
% PDF einbinden:  \pdfeinbinden{pfad/zur/datei.pdf}{Titel im Inhaltsverzeichnis}
\newcounter{pdfzaehler}
\newcommand{\pdfeinbinden}[2]{%
  \stepcounter{pdfzaehler}%
  \includepdf[
    pages=-,
    fitpaper=true,
    addtotoc={1,subsection,2,{#2},pdf\thepdfzaehler}
  ]{#1}%
}

% C-/Header-Datei einbinden:  \codeeinbinden{code/datei.c}
\newcommand{\codeeinbinden}[1]{%
  \subsection{\texorpdfstring{\texttt{\detokenize{#1}}}{#1}}%
  \lstinputlisting{#1}%
  \clearpage%
}
